✅ mcp 学习使用

给 AI Agent 的 mcp.json 配置：

```json
{
  "mcpServers": {
    "fetch": {
      "command": "uvx",
      "args": ["--with", "mcp\u003c2", "mcp-server-fetch"]
    },
    "nws_weather": {
      // 美国天气
      "command": "npx",
      "args": ["-y", "@h1deya/mcp-server-weather"]
    },
    "open_meteo": {
      // 所有的天气、空气质量、海洋天气、洪水风险、预测气候等
      "command": "npx",
      "args": ["-y", "-p", "open-meteo-mcp-server", "open-meteo-mcp-server"]
    }
  }
}
```
